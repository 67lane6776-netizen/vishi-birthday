    /* =================================
       SONG START POINT
       =================================

       Change this number if your MP3
       reaches "You can dance, you can jive..."
       at a slightly different time.

       Start with 46 seconds.
    */

    const DANCING_QUEEN_START = 46;



    /* =================================
       SCREEN FUNCTIONS
       ================================= */

    function hideAllScreens() {

      document
        .querySelectorAll(".screen")
        .forEach(screen => {

          screen.classList.remove("active");

        });

    }


    function showSad() {

      hideAllScreens();

      document
        .getElementById("sadScreen")
        .classList.add("active");

    }


    function showBirthday() {

      hideAllScreens();

      document
        .getElementById("birthdayScreen")
        .classList.add("active");


      /* 🎉 CONFETTI */

      createConfetti();


      /* 🎵 START DANCING QUEEN */

      const song =
        document.getElementById("birthdaySong");


      /*
        This makes the song jump directly
        to the "You can dance, you can jive..."
        section.
      */

      song.currentTime =
        DANCING_QUEEN_START;


      song.play().catch(() => {

        console.log(
          "Autoplay blocked. Tap DANCE QUEEN MODE."
        );

      });

    }



    function goBack() {

      hideAllScreens();

      document
        .getElementById("homeScreen")
        .classList.add("active");

    }



    /* =================================
       MUSIC BUTTON
       ================================= */

    function playMusic() {

      const song =
        document.getElementById("birthdaySong");


      if (song.paused) {

        /*
          If the song hasn't started,
          start at the Dancing Queen section.
        */

        if (song.currentTime < 1) {

          song.currentTime =
            DANCING_QUEEN_START;

        }

        song.play();

      } else {

        song.pause();

      }

    }



    /* =================================
       CONFETTI
       ================================= */

    function createConfetti() {

      const total = 180;


      for (
        let i = 0;
        i < total;
        i++
      ) {

        const confetti =
          document.createElement("div");


        confetti.classList.add(
          "confetti"
        );


        confetti.style.left =
          Math.random() * 100 + "vw";


        confetti.style.animationDuration =
          (Math.random() * 3 + 2) + "s";


        confetti.style.animationDelay =
          Math.random() * 1.5 + "s";


        confetti.style.background =
          randomColor();


        confetti.style.borderRadius =
          Math.random() > .5
            ? "50%"
            : "2px";


        document.body.appendChild(
          confetti
        );


        setTimeout(() => {

          confetti.remove();

        }, 6500);

      }

    }



    function randomColor() {

      const colors = [

        "#ff3eb5",
        "#ff8bdc",
        "#8c5cff",
        "#5ce1ff",
        "#ffffff",
        "#ffd84d",
        "#7fffd4"

      ];


      return colors[
        Math.floor(
          Math.random() * colors.length
        )
      ];

    }
